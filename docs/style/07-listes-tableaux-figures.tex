% Trois écrans par ligne, deux lignes tiennent sur une page.
\newcounter{plancheCol}

\newenvironment{planche}[2][]{%
  \def\plancheLabel{#1}%
  \def\plancheLegende{#2}%
  \begin{figure}[htbp]
    \centering
    \setlength\parskip{0pt}%
    \setcounter{plancheCol}{0}%
}{%
    \par\vspace{2pt}%
    \caption{\plancheLegende}%
    \if\relax\detokenize\expandafter{\plancheLabel}\relax\else\label{\plancheLabel}\fi
  \end{figure}%
}

\newcommand\maquette[3]{%
  \ifnum\value{plancheCol}>0 \hspace{0.075\linewidth}\fi
  \begin{minipage}[t]{0.28\linewidth}
    \centering
    \includegraphics[width=\linewidth]{#1}\par
    \vspace{5pt}%
    \pastille[1.6em]{#2}\hspace{0.4em}{\small\policeMoyenne #3}%
  \end{minipage}%
  \stepcounter{plancheCol}%
  \ifnum\value{plancheCol}=3
    \setcounter{plancheCol}{0}\par\vspace{12pt}%
  \fi
  \ignorespaces
}
