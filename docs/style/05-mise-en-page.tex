%  05 · Mise en page
%  En-tête = bandeau daté (projet · chapitre), pied = version · pagination.
% ---------------------------------------------------------------------

\geometry{
  a4paper,
  left=22mm, right=22mm,
  top=32mm, bottom=26mm,
  headheight=40pt, headsep=8mm,
  footskip=12mm,
}

\renewcommand\chaptermark[1]{\markboth{#1}{}}
\renewcommand\sectionmark[1]{}

\newcommand\piedDePage{%
  \textlabel[texte2]{\TypeDocument\ · v\VersionDoc}%
  \hfill
  {\policeChiffres\small
    \textcolor{texte}{\thepage}\textcolor{texte2}{\,/\,\pageref*{LastPage}}}%
}

\fancypagestyle{marche}{%
  \fancyhf{}%
  \renewcommand\headrulewidth{0pt}%
  \renewcommand\footrulewidth{0pt}%
  \fancyhead[L]{\bandeau{\NomAppli\ \NomAppliAccent}%
    {\ifnum\value{chapter}>0 \numchapitre\ · \fi\leftmark}}%
  \fancyfoot[L]{\piedDePage}%
}

% Pages d'ouverture de chapitre : pied seul
\fancypagestyle{plain}{%
  \fancyhf{}%
  \renewcommand\headrulewidth{0pt}%
  \renewcommand\footrulewidth{0pt}%
  \fancyfoot[L]{\piedDePage}%
}

\pagestyle{marche}
